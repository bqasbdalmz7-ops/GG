function shadeHex(x,f){var c=hexRGB(x);var r=c.map(function(v){return clamp(Math.round(v*f),0,255)});
 return 'rgb('+r[0]+','+r[1]+','+r[2]+')'}
function fillEll(g,cx,cy,rx,ry,c){g.fillStyle=c;
 for(var yy=-ry;yy<=ry;yy++){var t=yy/ry;var w=Math.floor(rx*Math.sqrt(Math.max(0,1-t*t)));
  if(w<=0)continue;g.fillRect(Math.round(cx-w),Math.round(cy+yy),w*2+1,1)}}
function orbS(g,cx,cy,rx,ry,base){
 fillEll(g,cx,cy,rx+1,ry+1,'#08060c');
 fillEll(g,cx,cy,rx,ry,base);
 fillEll(g,cx,cy-ry*0.30,rx*0.62,ry*0.42,shadeHex(base,1.34));
 fillEll(g,cx,cy+ry*0.62,rx*0.82,ry*0.30,shadeHex(base,0.40));
 g.fillStyle=shadeHex(base,1.85);
 g.fillRect(Math.round(cx-rx*0.55),Math.round(cy-ry+1),Math.max(1,Math.round(rx*0.5)),1);
 g.fillStyle=shadeHex(base,0.68);
 g.fillRect(Math.round(cx-rx),Math.round(cy),1,Math.round(ry*0.9));
}
function limbV(g,x1,y1,x2,y2,w1,w2,c){
 var n=Math.max(2,Math.round(Math.max(Math.abs(x2-x1),Math.abs(y2-y1))));
 for(var i=0;i<=n;i++){var t=i/n;var x=x1+(x2-x1)*t,y=y1+(y2-y1)*t,w=w1+(w2-w1)*t;
  g.fillStyle='#08060c';g.fillRect(Math.round(x-w-0.5),Math.round(y-w-0.5),2*Math.round(w)+2,2*Math.round(w)+2)}
 for(var i2=0;i2<=n;i2++){var t2=i2/n;var x3=x1+(x2-x1)*t2,y3=y1+(y2-y1)*t2,w3=w1+(w2-w1)*t2;
  g.fillStyle=c;g.fillRect(Math.round(x3-w3),Math.round(y3-w3),2*Math.round(w3)+1,2*Math.round(w3)+1)}
}
function plateV(g,x,y,w,h,c,r){
 r=r||2;fillEll(g,x+w/2,y+r,r,r,'#08060c');fillEll(g,x+w/2,y+h-r,r,r,'#08060c');
 g.fillStyle='#08060c';g.fillRect(x,y+r,w,h-2*r+1);
 fillEll(g,x+w/2,y+r,Math.max(0,r-1),Math.max(0,r-1),c);fillEll(g,x+w/2,y+h-r,Math.max(0,r-1),Math.max(0,r-1),c);
 g.fillStyle=c;g.fillRect(x+1,y+r,w-2,h-2*r+1);
 g.fillStyle=shadeHex(c,1.34);g.fillRect(x+1,y+1,w-2,1);
 g.fillStyle=shadeHex(c,0.68);g.fillRect(x+1,y+h-2,w-2,1);
}
function coneV(g,x,y,dir,len,w,c,sx,sy){
 sx=sx||1;sy=sy||1;
 g.fillStyle='#08060c';g.beginPath();g.moveTo(x+sx,y);g.lineTo(x+Math.cos(dir)*(len+1)*sx-w-1,y+Math.sin(dir)*(len+1)*sy-w-1);g.lineTo(x+Math.cos(dir)*(len+1)*sx+w+1,y+Math.sin(dir)*(len+1)*sy+w+1);g.fill();
 g.fillStyle=c;g.beginPath();g.moveTo(x+sx,y);g.lineTo(x+Math.cos(dir)*(len+1)*sx-w,y+Math.sin(dir)*(len+1)*sy-w);g.lineTo(x+Math.cos(dir)*(len+1)*sx+w,y+Math.sin(dir)*(len+1)*sy+w);g.fill();
}
function whiteVer(c){
 var w=C(c.width,c.height),g=w.getContext('2d');
 g.drawImage(c,0,0);g.globalCompositeOperation='source-in';
 g.fillStyle='#ffffff';g.fillRect(0,0,w.width,w.height);return w;
}
function bakeCanvas(w,h,painter){
 var c=C(w,h),g=c.getContext('2d');g.imageSmoothingEnabled=false;
 g.save();g.translate(w/2,h/2);painter(g);g.restore();
 return {c:c,wf:whiteVer(c)};
}
function paintHero(g,ch,f){
 var F=ch.flags,b=shadeHex(ch.col,0.9);
 var run=f>=2,ph=run?(f-2):f*0.5;
 var bob=run?(Math.abs(Math.sin(ph*Math.PI))*1.5):(f%2?0.5:0);
 var legA=run?Math.sin(ph*Math.PI*2)*2.5:0;
 var lean=F.dual?1:0.5;
 var B='#1a1428',E='#08060c';
 var skin='#e8b98a',clo='#2b2038';
 if(ch.id==='mage')clo='#3a2a5e';if(ch.id==='knight')clo='#4a5568';
 if(ch.id==='assassin')clo='#20242e';if(ch.id==='sniper')clo='#3a3d2e';
 if(ch.id==='berserk')clo=null;
 if(F.cloak){g.globalAlpha=0.95;fillEll(g,-2-bob*0.3,3+bob,6,8,shadeHex(clo,0.6));
  g.fillStyle=shadeHex(clo,0.68);g.fillRect(-7,3+bob,3,8+bob);g.globalAlpha=1}
 limbV(g,-2-legA,8, -4-legA,13+bob*0.3,1.6,1.2,b);
 limbV(g, 2+legA,8,  4+legA,13+bob*0.3,1.6,1.2,b);
 if(F.armor){plateV(g,-6,-1+bob,12,9,'#4a5568',3)}
