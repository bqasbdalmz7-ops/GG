function nearestTarget(range){
 var best=null,bd=range*range;
 for(var i=0;i<R.en.length;i++){var e=R.en[i];
  if(e.dying)continue;if(e.ty==='ghost'&&e.ph===0)continue;
  var dx=e.x-R.px,dy=e.y-R.py,d2=dx*dx+dy*dy;
  if(d2<bd){bd=d2;best=e}}
 for(var bi=0;bi<R.bosses.length;bi++){var B=R.bosses[bi];
  var dx2=B.x-R.px,dy2=B.y-R.py,d22=dx2*dx2+dy2*dy2;
  if(d22<bd){bd=d22;best=B}}
 return best;
}

function updBullets(dt){
 for(var i=R.bs.length-1;i>=0;i--){
  var b=R.bs[i];
  if(b.home>0&&R.en.length){
   var tg=nearestTarget(400);
   if(tg){var want=Math.atan2(tg.y-b.y,tg.x-b.x),cur=Math.atan2(b.vy,b.vx);
    var dd=angDiff(want,cur);cur+=clamp(dd,-b.home*dt,b.home*dt)*b.home;
    var sp=Math.hypot(b.vx,b.vy);b.vx=Math.cos(cur)*sp;b.vy=Math.sin(cur)*sp}}
  b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
  if(Math.random()<0.35)R.parts.push({x:b.x,y:b.y,vx:0,vy:0,life:0.18,r:2,col:b.trail,glow:1});
  var hitSomething=false;
  for(var j=0;j<R.en.length;j++){var e=R.en[j];
   if(e.dying||b.hit[e.ty+j])continue;
   if(e.ty==='ghost'&&e.ph===0)continue;
   var rr2=e.r+b.r;
   if(segHit(b.x-b.vx*dt,b.y-b.vy*dt,b.x,b.y,e.x,e.y,rr2)){
    b.hit[e.ty+j]=1;hitEnemy(e,b.dmg,b);
    if(e.boss)continue;
    if(b.kind==='vial')spawnCloud(b.x,b.y);
    else if(b.splash)splashDmg(b.x,b.y,b.splash,b.dmg*0.6);
    b.pierce--;hitSomething=true;
    if(b.pierce<0){R.bs.splice(i,1);break}}}
  for(var bi2=0;bi2<R.bosses.length;bi2++){var bb=R.bosses[bi2];
   if(b.hit['B'+bi2])continue;
   var rrb=bb.r+b.r;
   if(segHit(b.x-b.vx*dt,b.y-b.vy*dt,b.x,b.y,bb.x,bb.y,rrb)){b.hit['B'+bi2]=1;hitBoss(bb,b.dmg);

    else if(b.splash)splashDmg(b.x,b.y,b.splash,b.dmg*0.6);
    b.pierce--;hitSomething=true;
    if(b.pierce<0){R.bs.splice(i,1);break}}}
  if(hitSomething)continue;
  if(b.life<=0){
   if(b.kind==='vial')spawnCloud(b.x,b.y);
   else if(b.splash&&!hitSomething&&b.kind==='fire')splashDmg(b.x,b.y,b.splash,b.dmg*0.6);
   R.bs.splice(i,1)}
 }
